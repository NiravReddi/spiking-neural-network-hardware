class ciphar10_model(nn.Module):
    def __init__(self, input_shape: int, hidden_units: int, output_shape: int):
        super().__init__()
        self.block_1 = nn.Sequential(
        nn.Conv2d(in_channels=input_shape, 
                      out_channels=hidden_units, 
                      kernel_size=3, # how big is the square that's going over the image?
                      stride=1, # default
                      padding=1),# options = "valid" (no padding) or "same" (output has same shape as input) or int for specific number 
            nn.ReLU(),
            nn.Conv2d(in_channels=hidden_units, 
                      out_channels=hidden_units*10,
                      kernel_size=3,
                      stride=1,
                      padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2,
                         stride=2) # default stride value is same as kernel_size
        )
        self.block_2 = nn.Sequential(
            nn.Conv2d(hidden_units*10, hidden_units*10, 3, padding=1),
            nn.ReLU(),
            nn.Conv2d(hidden_units*10, hidden_units, 3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2)
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),
            # Where did this in_features shape come from? 
            # It's because each layer of our network compresses and changes the shape of our inputs data.
            nn.Linear(in_features=hidden_units*7*7, 
                      out_features=output_shape)
        )
    def forward(self, x: torch.Tensor):
        x = self.block_1(x)
        ##print(x.shape)
        x = self.block_2(x)
        #print(x.shape)
        x = self.classifier(x)
        #print(x.shape)
        return x

from tqdm.auto import tqdm

model_2.to("cuda")
test_loss_l=[]
test_accuracy_l=[]
train_loss_l=[]
train_accuracy_l=[]
index=[]
c=0

epochs = 15

for epoch in tqdm(range(epochs)):
    print("epoch : ",epoch,end=" ")
    
    train_loss=0
    train_acc = 0
    
    for batch, (X,y) in enumerate(train_dataloader):
        X,y=X.to("cuda"),y.to("cuda")
        model_2.train()
        y_pred = model_2(X)
       
        
        loss = loss_fun(y_pred,y)
        train_loss += loss
        
        optimizer.zero_grad()
        
        loss.backward()
        train_acc += accuracy_fun(y_pred,y)
        optimizer.step()
    
    train_loss /= len(train_dataloader)
    train_acc /= len(train_dataloader)
    
    # testing
    test_loss=0
    test_acc = 0
    with torch.inference_mode():
        for X_test,y_test in test_dataloader:
            X_test,y_test=X_test.to("cuda"),y_test.to("cuda")
            test_pred =model_2(X_test)
            
            test_loss += loss_fun(test_pred,y_test)
            
            test_acc += accuracy_fun(test_pred,y_test) 
        test_loss /= len(test_dataloader)
        test_acc /= len(test_dataloader)
    print("Training loss :", float(train_loss),"Test Loss : ", float(test_loss),"Test Accuracy :",float(test_acc))
    train_loss_l.append(float(train_loss)*100)
    train_accuracy_l.append(float(train_acc)*100)
    test_loss_l.append(float(test_loss)*100)
    test_accuracy_l.append(float(test_acc)*100)
    index.append(c)
    c+=1
